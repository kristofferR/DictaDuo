cmake_minimum_required(VERSION 3.24)
project(SottoDuoLinuxGUI VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Quick QuickControls2 Widgets Network Svg DBus)
find_package(LayerShellQt 6.6 REQUIRED)
find_package(PkgConfig QUIET)
if(PkgConfig_FOUND)
  pkg_check_modules(LIBPORTAL QUIET IMPORTED_TARGET "libportal>=0.11")
endif()
qt_standard_project_setup(REQUIRES 6.8)
qt_add_executable(sottoduo-gui main.cpp Bridge.cpp Bridge.h HudSurface.cpp HudSurface.h DesktopIntegration.cpp DesktopIntegration.h GuiInstance.cpp GuiInstance.h PortalShortcuts.cpp PortalShortcuts.h)
qt_add_qml_module(sottoduo-gui URI SottoDuo VERSION 1.0 QML_FILES
  qml/Main.qml qml/SButton.qml qml/SLabel.qml qml/StatusDot.qml qml/Group.qml qml/Setting.qml qml/Dictation.qml qml/MicrophoneTestButton.qml
  qml/History.qml qml/HistoryDetail.qml qml/HistoryChip.qml qml/Microphone.qml qml/Preferences.qml qml/DictionaryEditor.qml qml/Computer.qml qml/Hud.qml qml/DjiSettings.qml qml/LevelMeter.qml qml/ShortcutSettings.qml qml/PortalShortcutSettings.qml qml/ConnectionSettings.qml
  RESOURCES mark.svg preview.json)
target_link_libraries(sottoduo-gui PRIVATE Qt6::Quick Qt6::QuickControls2 Qt6::Widgets Qt6::Network Qt6::Svg Qt6::DBus LayerShellQt::Interface)
if(LIBPORTAL_FOUND)
  target_link_libraries(sottoduo-gui PRIVATE PkgConfig::LIBPORTAL)
  target_compile_definitions(sottoduo-gui PRIVATE SOTTODUO_HAS_LIBPORTAL)
endif()
target_compile_options(sottoduo-gui PRIVATE -Wall -Wextra -Werror)
install(TARGETS sottoduo-gui RUNTIME DESTINATION bin)
install(PROGRAMS "${CMAKE_CURRENT_SOURCE_DIR}/../../../build/linux-client/sottoduo"
                 "${CMAKE_CURRENT_SOURCE_DIR}/../../../build/linux-client/sottoduo-destination"
        DESTINATION bin)
install(FILES sottoduo.desktop DESTINATION share/applications)
install(FILES mark.svg DESTINATION share/icons/hicolor/scalable/apps RENAME sottoduo.svg)

include(CTest)
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  qt_add_executable(sottoduo-gui-tests tests/BridgeTest.cpp Bridge.cpp Bridge.h HudSurface.cpp HudSurface.h DesktopIntegration.cpp DesktopIntegration.h)
  qt_add_resources(sottoduo-gui-tests "fixtures" PREFIX "/qt/qml/SottoDuo" FILES preview.json)
  target_link_libraries(sottoduo-gui-tests PRIVATE Qt6::Quick Qt6::QuickControls2 Qt6::Widgets Qt6::Network Qt6::Svg Qt6::Test LayerShellQt::Interface)
  target_compile_definitions(sottoduo-gui-tests PRIVATE SOTTODUO_QML_DIR="${CMAKE_CURRENT_SOURCE_DIR}/qml")
  target_compile_options(sottoduo-gui-tests PRIVATE -Wall -Wextra -Werror)
  add_test(NAME linux-gui COMMAND sottoduo-gui-tests)
  set_tests_properties(linux-gui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
  find_program(DBUS_RUN_SESSION dbus-run-session REQUIRED)
  qt_add_executable(sottoduo-desktop-tests tests/DesktopTest.cpp DesktopIntegration.cpp DesktopIntegration.h GuiInstance.cpp GuiInstance.h)
  target_link_libraries(sottoduo-desktop-tests PRIVATE Qt6::Core Qt6::DBus Qt6::Test)
  target_compile_definitions(sottoduo-desktop-tests PRIVATE SOTTODUO_GUI_EXECUTABLE="$<TARGET_FILE:sottoduo-gui>")
  target_compile_options(sottoduo-desktop-tests PRIVATE -Wall -Wextra -Werror)
  add_dependencies(sottoduo-desktop-tests sottoduo-gui)
  add_test(NAME linux-desktop COMMAND "${DBUS_RUN_SESSION}" -- $<TARGET_FILE:sottoduo-desktop-tests>)
  set_tests_properties(linux-desktop PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
endif()
